# CSAT (Customer Satisfaction) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** When a ticket is finalized, show the customer an in-app popup to rate the service (1–5 stars + optional comment); store each rating immutably attributed to the last assigned agent; expose results to Admins in-app and via a REST API (raw + per-agent aggregates).

**Architecture:** New `Ticket::SatisfactionRating` ActiveRecord model (mirrors the `RecentClose` convention). The customer rates via a Vue desktop popup triggered by the existing `ticketUpdates` subscription (state → closed) + an on-open fallback, submitting through a GraphQL mutation. Admin display via a permission-gated GraphQL field. External pull via REST controllers (`/api/v1/csat/surveys`, `/api/v1/csat/stats`). No email, no external page, no public token.

**Tech Stack:** Ruby on Rails, GraphQL (graphql-ruby), Pundit, PostgreSQL; Vue 3 + TypeScript + Apollo + FormKit (desktop app); RSpec + FactoryBot; Vitest + Testing Library.

**Spec:** `docs/superpowers/specs/2026-06-09-csat-design.md`

**Canonical names (use EXACTLY these across all tasks):**
- Model `Ticket::SatisfactionRating` → table `ticket_satisfaction_ratings`
- Columns: `ticket_id`, `customer_id`, `agent_id`(null), `group_id`(null), `score`(int 1..5), `comment`(text null), `created_by_id`/`updated_by_id`, `created_at`/`updated_at`
- Permissions: `csat.read` (admin display + API), `admin.csat` (settings edit) — both granted to Admin role only
- Settings: `csat_integration`(boolean, default `false`, frontend), `csat_comment`(select `off`/`optional`/`required`, default `optional`, frontend), `csat_closed_state_types`(array, default `['closed']`, backend)
- GraphQL: type `Gql::Types::Ticket::SatisfactionRatingType`; `TicketType` fields `satisfaction` + `satisfactionRatable`; mutation `Gql::Mutations::Ticket::SatisfactionRating::Create` → `ticketSatisfactionRatingCreate`; input `Gql::Types::Input::Ticket::SatisfactionRatingInputType`; policy `Ticket::SatisfactionRatingPolicy`
- REST: `CsatSurveysController#index` (`/api/v1/csat/surveys`), `CsatStatsController#index` (`/api/v1/csat/stats`); controller policies `Controllers::CsatSurveysControllerPolicy` / `Controllers::CsatStatsControllerPolicy`; aggregation service `Service::Csat::Stats`
- Frontend (desktop): popup `TicketSatisfactionDialog.vue`; composable `useTicketSatisfaction`; **reuse existing** `rating` FormKit field (`apps/desktop/components/Form/fields/FieldRating`); admin panel `TicketSatisfactionInfo`

```ruby
# app/models/ticket/satisfaction_rating.rb
class Ticket::SatisfactionRating < ApplicationModel
  include HasDefaultModelUserRelations

  belongs_to :ticket
  belongs_to :customer, class_name: 'User'
  belongs_to :agent,    class_name: 'User', optional: true
  belongs_to :group,    optional: true

  validates :score, presence: true, inclusion: { in: 1..5 }
  validates :ticket_id, uniqueness: { scope: :customer_id }

  # write-once: never mutate after registration
  attr_readonly :ticket_id, :customer_id, :agent_id, :score

  before_create :snapshot_agent_and_group

  private

  def snapshot_agent_and_group
    self.group_id ||= ticket.group_id
    self.agent_id ||= resolve_last_agent
  end

  # last assigned agent at registration time; fall back to the last human owner in history
  def resolve_last_agent
    return ticket.owner_id if ticket.owner_id.present? && ticket.owner_id != 1

    last = ticket.history_get
                 .select { |h| h['attribute'] == 'owner' && h['value_from'].present? }
                 .map { |h| h['id_to'] || h['value_from'] }
                 .compact
    User.where(id: last).where.not(id: 1).last&.id
  end
end
```

> Note: `history_get` returns the ticket's History entries (Zammad core). If the owner-change shape differs in practice, adjust the filter in `resolve_last_agent`; the spec's history case pins the expected behavior.

```ruby
# app/policies/ticket/satisfaction_rating_policy.rb
class Ticket::SatisfactionRatingPolicy < ApplicationPolicy
  def create?
    return not_authorized(__('CSAT is not enabled')) if !Setting.get('csat_integration')
    return not_authorized(__('Only the ticket customer can rate')) if user.id != record.ticket.customer_id
    return not_authorized(__('Ticket is not finalized')) if !closed_state?
    return not_authorized(__('This ticket was already rated')) if already_rated?

    true
  end

  def show?
    return false if !Setting.get('csat_integration')
    return true  if user.id == record.customer_id

    user.permissions?('csat.read')
  end

  private

  def closed_state?
    Setting.get('csat_closed_state_types').include?(record.ticket.state.state_type.name)
  end

  def already_rated?
    Ticket::SatisfactionRating
      .where(ticket_id: record.ticket_id, customer_id: record.ticket.customer_id)
      .where.not(id: record.id)
      .exists?
  end
end
```

```ruby
# app/graphql/gql/mutations/ticket/satisfaction_rating/create.rb
module Gql::Mutations
  class Ticket::SatisfactionRating::Create < BaseMutation
    description 'Create a customer satisfaction rating for a ticket.'

    argument :input, Gql::Types::Input::Ticket::SatisfactionRatingInputType, description: 'The rating data'

    field :satisfaction_rating, Gql::Types::Ticket::SatisfactionRatingType, description: 'The created rating.'

    def resolve(input:)
      rating = ::Ticket::SatisfactionRating.new(
        ticket:   input[:ticket_id],
        customer: context.current_user,
        score:    input[:score],
        comment:  comment_value(input[:comment]),
      )

      Pundit.authorize(context.current_user, rating, :create?)
      rating.save!

      { satisfaction_rating: rating }
    rescue ActiveRecord::RecordInvalid => e
      error_response({ message: e.message })
    end

    private

    def comment_value(comment)
      return comment if Setting.get('csat_comment') != 'off'

      nil
    end
  end
end
```

> `Pundit.authorize` raises `Pundit::NotAuthorizedError` on failure, which `BaseMutation`/`HandlesAuthorization` converts to `Exceptions::Forbidden` (asserted in the spec). The `input[:ticket_id]` argument is already resolved to a `Ticket` record by `loads:` in the input type.

```ruby
# app/services/service/csat/stats.rb
class Service::Csat::Stats
  def initialize(params)
    @params = params
  end

  def execute
    rel = filtered_scope
    {
      overall:  aggregate(rel),
      by_agent: by_agent(rel),
    }
  end

  private

  def filtered_scope
    rel = Ticket::SatisfactionRating.all
    rel = rel.where(agent_id: @params[:agent_id]) if @params[:agent_id].present?
    rel = rel.where(group_id: @params[:group_id]) if @params[:group_id].present?
    rel = rel.where(created_at: Time.zone.parse(@params[:date_from])..) if @params[:date_from].present?
    rel = rel.where(created_at: ..Time.zone.parse(@params[:date_to]))   if @params[:date_to].present?
    rel
  end

  def aggregate(rel)
    scores = rel.pluck(:score)
    {
      count:         scores.size,
      average:       scores.empty? ? nil : (scores.sum.to_f / scores.size).round(2),
      distribution:  (1..5).index_with { |s| scores.count(s) },
      response_rate: response_rate,
    }
  end

  def by_agent(rel)
    rel.group(:agent_id).pluck(:agent_id, Arel.sql('COUNT(*)'), Arel.sql('AVG(score)')).map do |agent_id, count, avg|
      {
        agent_id:,
        agent:    User.find_by(id: agent_id)&.fullname,
        count:    count,
        average:  avg.nil? ? nil : avg.to_f.round(2),
      }
    end
  end

  # Taxa de Resposta (geral): dos tickets finalizados na janela (close_at), quantos têm Avaliação.
  # close_at é populado por Ticket::SetsCloseTime quando o ticket entra em estado 'closed'.
  def response_rate
    cohort = finalized_cohort
    total  = cohort.count
    return nil if total.zero?

    rated = cohort.where(id: Ticket::SatisfactionRating.select(:ticket_id)).count
    (rated.to_f / total).round(2)
  end

  # coorte = tickets cujo close_at cai na janela, + filtros de grupo/atendente (owner)
  def finalized_cohort
    rel = Ticket.where.not(close_at: nil)
    rel = rel.where(close_at: Time.zone.parse(@params[:date_from])..) if @params[:date_from].present?
    rel = rel.where(close_at: ..Time.zone.parse(@params[:date_to]))   if @params[:date_to].present?
    rel = rel.where(group_id: @params[:group_id]) if @params[:group_id].present?
    rel = rel.where(owner_id: @params[:agent_id]) if @params[:agent_id].present?
    rel
  end
end
```

> **Taxa de Resposta** = tickets finalizados na janela (`close_at`) que têm Avaliação ÷ tickets finalizados na janela. `close_at` é populado por `Ticket::SetsCloseTime` ao entrar em estado `closed`. Métrica **geral** (não vai no `by_agent`). Documentar a definição no doc da API. Ver `docs/csat/CONTEXT.md`.

```ts
// useTicketSatisfaction.ts
import { watch } from 'vue'

import { useDialog } from '#desktop/components/CommonDialog/useDialog.ts'
import { useApplicationStore } from '#shared/stores/application.ts'
import { MutationHandler } from '#shared/server/apollo/handler/index.ts'
import { useTicketSatisfactionRatingCreateMutation } from '../graphql/mutations/ticketSatisfactionRatingCreate.api.ts'
import type { TicketById } from '#shared/entities/ticket/types.ts'
import type { Ref } from 'vue'

const dismissedKey = (id: string) => `csat-dismissed-${id}`

export const useTicketSatisfaction = (
  ticket: Ref<TicketById | undefined>,
  refetchTicket: () => void,
) => {
  const { config } = useApplicationStore()

  const { open } = useDialog({
    name: 'ticket-satisfaction',
    global: true,
    component: () =>
      import('../components/TicketDetailView/TicketSatisfaction/TicketSatisfactionDialog.vue'),
  })

  const createMutation = new MutationHandler(useTicketSatisfactionRatingCreateMutation())

  const openDialog = (ticketId: string) => {
    open({
      onSubmit: async (data: { score: number; comment?: string }) => {
        await createMutation.send({ input: { ticketId, score: data.score, comment: data.comment } })
        refetchTicket() // criar rating não mexe no Ticket → ticketUpdates não dispara; refetch atualiza satisfactionRatable + satisfaction
      },
      onDismiss: () => localStorage.setItem(dismissedKey(ticketId), 'true'),
    })
  }

  watch(
    () => ticket.value?.satisfactionRatable,
    (ratable) => {
      if (!config.csat_integration || !ratable || !ticket.value) return
      if (localStorage.getItem(dismissedKey(ticket.value.id))) return
      openDialog(ticket.value.id)
    },
    { immediate: true },
  )

  return { openSatisfactionDialog: () => ticket.value && openDialog(ticket.value.id) }
}
```

> `satisfactionRatable` is recomputed server-side and arrives via the existing `ticketUpdates` subscription, so the `watch` fires when the agent closes the ticket live, and on mount when already closed.
